C4Container
    title Container diagram: Message Vault from source
    System_Ext(backups, "Phone backup files", "On the user's computer")
    Person(user, "User", "A user with messages to back up")
    System_Boundary(vault, "Message Vault") {
        Container(desktop, "Desktop App", "Tauri, React, src-tauri/ and web/", "Native window. Extract and Format write files on disk. Import and browse call the API.")
        Container(spa, "Webpage", "React + Vite, web/", "Same SPA in the browser at http://127.0.0.1:8080 and inside the Tauri WebView")
        Container(server, "Vault Server", "Rust, Axum", "cargo run -p message-vault-server -- serve. Serves static/ and /v1 on 127.0.0.1:8080")
        ContainerDb(db, "Database", "SQLite", "data/vault.db. Attachment files under data/. Not a separate process.")
    }

    Rel(user, desktop, "Uses", "https")
    Rel(user, spa, "Uses", "https")
    Rel(desktop, spa, "Embeds in WebView")
    Rel(desktop, backups, "Reads during Extract")
    Rel(desktop, server, "HTTP /v1")
    Rel_U(spa, server, "GET / and /v1")
    Rel_D(server, db, "Reads and writes")

            Container(vite, "Vite", "localhost:5173", "Serves web/ live")
        }
        Deployment_Node(backend, "Message Vault Backend", "Processes and filesystems") {
            Deployment_Node(srv, "Server process", "127.0.0.1:8080") {
                Container(api, "API", "HTTP /v1")
                Container(web, "Web", "HTTP GET /", "Webpage from web/")
            }
            Deployment_Node(disk, "Filesystem", "Repo working directory") {
                Container(webDir, "web/", "React + Vite source", "SPA on disk")
                ContainerDb(db, "Database", "SQLite", "data/vault.db")
                Container(dataDir, "data/", "Attachment files")
            }
        }
    }
    Rel(spa, web, "GET /")
    Rel(spa, api, "/v1")
    Rel(desktop, vite, "Loads UI")
    Rel(desktop, api, "/v1 data")
    Rel(vite, webDir, "Serves")
    Rel(web, webDir, "GET / from")
    Rel(api, db, "Reads and writes")
    Rel(api, dataDir, "Assets")


sequenceDiagram
    autonumber
    actor Dev as Developer

    box Local Machine
        participant App as Desktop App (Tauri)
        participant Vite as Vite (port 5173)
        participant Repo as git repo (message-vault)
    end

    box Docker Containers
        participant Vault as Vault (port 8080)
        participant SQLite as SQLite Browser (port 8081)
    end

    box Docker Volumes
        participant AppMount as /app (source mount)
        participant Target as /app/target (Rust cache)
        participant Libs as Cargo registry cache
        participant Data as /app/data (SQLite DB)
    end

    Dev->>Repo: Edits Rust or frontend files
    Note over Repo,AppMount: Repo mounted as /app in Vault container

    Dev->>App: cargo tauri dev
    App->>Vite: Starts Vite dev server
    App->>App: Loads http://localhost:5173

    loop Page open / refresh
        Vite-->>App: Serves web/ assets
    end

    App->>Vault: Login (port 8080)
    Vault-->>App: Auth response

    loop Messages
        App->>Vault: Fetch / send messages
        Vault-->>App: Message data
        Vault->>Data: Persists to SQLite
    end

    Dev->>Vault: Restart container (recompiles)
    Vault->>AppMount: Reads source from /app
    Vault->>Target: Uses cached build artifacts
    Vault->>Libs: Uses cached dependencies
    Vault-->>Dev: Ready (port 8080)

    Dev->>SQLite: Opens browser (port 8081)
    SQLite->>Data: Reads shared database file
    SQLite-->>Dev: Shows database contents
